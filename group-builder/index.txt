function genId() {
  return Math.random().toString(36).substr(2, 9);
}

function rankIdx(rank) {
  const i = RANKS.indexOf(rank);
  return i; // -1 if unknown
}

function isBlackBelt(rank) {
  return BLACK_BELT_RANKS.has(rank);
}

function isAdult(age) {
  return age >= 18;
}

function rankBandFor(rank) {
  const idx = rankIdx(rank);
  if (idx < 0) return null;
  return RANK_BANDS.find(b => idx >= b.minIdx && idx <= b.maxIdx) || null;
}

function rankWindowAllowed(groupRanks) {
  if (!groupRanks.length) return true;

  const idxs = groupRanks.map(rankIdx).filter(i => i >= 0);
  if (!idxs.length) return true;

  const minIdx = Math.min(...idxs);
  const maxIdx = Math.max(...idxs);
  const hasGup = idxs.some(i => i >= 1 && i <= 10);
  const hasBlackBelt = idxs.some(i => i >= 11);
  const memberCount = groupRanks.length;

  if (hasBlackBelt && !hasGup) {
    return maxIdx - minIdx <= 3;
  }

  if (hasGup && !hasBlackBelt) {
    if (memberCount >= 6) return maxIdx - minIdx <= 5;
    if (memberCount >= 4) return maxIdx - minIdx <= 4;
    return maxIdx - minIdx <= 2;
  }

  return maxIdx - minIdx <= 2;
}

function normalizeGender(value) {
  const v = String(value || '').trim().toLowerCase();
  if (!v) return 'Unknown';
  if (v.startsWith('m')) return 'Male';
  if (v.startsWith('f')) return 'Female';
  return capitalizeWords(value.trim());
}

  return ADULT_BRACKETS.find(b => age >= b.min && age <= b.max) || null;
}

function ageSpanLimitFor(age) {
  if (age < 14) return 3;
  if (age < 38) return 4;
  return 35;
}

function ageSpanFlexLimitFor(age) {
  if (age < 14) return 5;
  if (age < 38) return 14;
  return 35;
}

function recommendedMaxSize(totalCompetitors) {
  if (totalCompetitors <= 120) return 6;
  if (totalCompetitors <= 250) return 7;
  return 13;
}

  const maxSize = recommendedMaxSize(totalCompetitors);
  const minInput = document.getElementById('minSize');

function groupGender(comps) {
  if (!comps.length) return 'Unknown';
  const genders = [...new Set(comps.map(c => c.gender || 'Unknown'))];
  return genders.length === 1 ? genders[0] : 'Mixed';
}

function groupRankBand(comps) {
  if (!comps.length) return 'Unknown Band';
  const bands = [...new Set(comps.map(c => {
    const band = rankBandFor(c.rank);
    return band ? band.label : 'Unknown Band';
  }))];
  return bands.length === 1 ? bands[0] : 'Mixed Bands';
}

function isTtldRank(rank) {
  return String(rank || '').trim() === 'TTLD';
}

function groupIsTtldOnly(group) {
  return !!group && group.competitors.length > 0 && group.competitors.every((c) => isTtldRank(c.rank));
}

function groupAgeRange(comps) {
  if (!comps.length) return 'No ages';
  const ages = comps.map(c => c.age);
  const lo = Math.min(...ages), hi = Math.max(...ages);
  return lo === hi ? `Age ${lo}` : `Ages ${lo}–${hi}`;
}

function groupRankRange(comps) {
  if (!comps.length) return 'No ranks';
  const idxs = comps.map(c => rankIdx(c.rank)).filter(i => i >= 0);
  if (!idxs.length) return 'Unknown Rank';
  const lo = Math.min(...idxs), hi = Math.max(...idxs);
  return lo === hi ? RANKS[lo] : `${RANKS[lo]} – ${RANKS[hi]}`;
}

function autoName(group) {
  if (!group.competitors.length) return 'Empty Group';
  const gender = group.gender || groupGender(group.competitors);
  const rankRange = groupRankRange(group.competitors);
  return `${gender} ${rankRange} · ${groupAgeRange(group.competitors)}`;
}

function syncGroupDisplayMetadata(group) {
  if (!group) return;

  if (!group.competitors.length) {
    group.gender = 'Unknown';
    group.rankBand = 'Unknown Band';
    group.isAdult = false;
    group.name = 'Empty Group';
    return;
  }

  group.gender = groupGender(group.competitors);
  group.rankBand = groupRankBand(group.competitors);
  group.isAdult = group.competitors.some((c) => isAdult(c.age));
  group.name = autoName(group);
}

function capitalizeWords(str) {

}

    .replace(/&/g, '&amp;')
    .replace(/</g, '&lt;')
    .replace(/>/g, '&gt;')
    .replace(/"/g, '&quot;');
}

function parseCSVLine(line) {
  const result = [];
  let cur = '', inQ = false;
  for (let i = 0; i < line.length; i++) {
    const ch = line[i];
    if (ch === '"') { inQ = !inQ; }
    else if (ch === ',' && !inQ) { result.push(cur.trim()); cur = ''; }
    else { cur += ch; }
  }
  result.push(cur.trim());
  return result;
}

function normalizeImportHeader(value) {
  return String(value || '')
   .replace(/^\uFEFF/, '')
   .trim()
   .toLowerCase()
   .replace(/\s+/g, ' ');
}

function normalizeImportKey(value) {
  return String(value || '')
   .trim()
   .toLowerCase()
   .replace(/\s+/g, '');
}

function createImportError(row, field, issue, value) {
  return {
   row,
   field,
   issue,
   value: value == null ? '' : String(value),
  };
}

function parseDob(value, row, errors) {
  const raw = String(value || '').trim();
  const match = raw.match(/^(\d{1,2})\/(\d{1,2})\/(\d{4})$/);
  if (!match) {
   errors.push(createImportError(row, 'DOB', 'Must use MM/DD/YYYY.', raw));
   return null;
  }

  const month = Number.parseInt(match[1], 10);
  const day = Number.parseInt(match[2], 10);
  const year = Number.parseInt(match[3], 10);
  const dob = new Date(year, month - 1, day);

  if (
   dob.getFullYear() !== year ||
   dob.getMonth() !== month - 1 ||
   dob.getDate() !== day
  ) {
   errors.push(createImportError(row, 'DOB', 'Is not a valid calendar date.', raw));
   return null;
  }

  const today = new Date();
  const age = today.getFullYear() - year - (
   today.getMonth() < month - 1 ||
   (today.getMonth() === month - 1 && today.getDate() < day)
  ? 1 : 0);

  if (age < 0) {
   errors.push(createImportError(row, 'DOB', 'Cannot be in the future.', raw));
   return null;
  }

  return {
   raw,
   iso: `${String(year).padStart(4, '0')}-${String(month).padStart(2, '0')}-${String(day).padStart(2, '0')}`,
   age,
  };
}

function validateRequiredText(value, label, maxLength, row, errors) {
  const text = String(value || '').trim();
  if (!text) {
   errors.push(createImportError(row, label, 'Is required.', text));
   return null;
  }
  if (text.length > maxLength) {
   errors.push(createImportError(row, label, `Must be ${maxLength} characters or fewer.`, text));
   return null;
  }
  return text;
}

function validateOptionalText(value, label, maxLength, row, errors) {
  const text = String(value || '').trim();
  if (!text) {
   return '';
  }
  if (text.length > maxLength) {
   errors.push(createImportError(row, label, `Must be ${maxLength} characters or fewer.`, text));
   return null;
  }
  return text;
}

function validateEmail(value, row, errors) {
  const text = String(value || '').trim();
  if (!text) {
   errors.push(createImportError(row, 'Email', 'Is required.', text));
   return null;
  }
  if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(text)) {
   errors.push(createImportError(row, 'Email', 'Must be a valid email address.', text));
   return null;
  }
  return text;
}

function validatePhone(value, row, errors) {
  const text = String(value || '').trim();
  if (!text) {
   return '';
  }
  if (!/^\d{10}$/.test(text)) {
   errors.push(createImportError(row, 'Phone', 'Must use xxxyyyzzzz format.', text));
   return null;
  }
  return text;
}

function validateAssociationNumber(value, row, errors) {
  const text = String(value || '').trim().toUpperCase();
  if (!text) {
   errors.push(createImportError(row, 'Association #', 'Is required.', text));
   return null;
  }
  if (!/^(?:D\d{5}|G\d{6})$/.test(text)) {
   errors.push(createImportError(row, 'Association #', 'Must be D##### or G######.', text));
   return null;
  }
  return text;
}

function validateGender(value, row, errors) {
  const text = String(value || '').trim().toLowerCase();
  if (!text) {
   errors.push(createImportError(row, 'Gender', 'Is required.', value));
   return null;
  }
  if (text !== 'm' && text !== 'male' && text !== 'f' && text !== 'female') {
   errors.push(createImportError(row, 'Gender', 'Must be M or F.', value));
   return null;
  }
  return text.startsWith('m') ? 'Male' : 'Female';
}

function validateRank(value, row, errors) {
  const key = String(value || '').trim().toUpperCase().replace(/\s+/g, '');
  const normalized = IMPORT_RANK_MAP[key.toLowerCase()];
  if (!normalized) {
   errors.push(createImportError(row, 'Current Rank', 'Must be TTLD, G10-G1, CDB, or D1-D3.', value));
   return null;
  }
  return normalized;
}

function validateHeight(value, row, errors) {
  const text = String(value || '').trim();
  const height = Number.parseInt(text, 10);
  if (!Number.isFinite(height)) {
   errors.push(createImportError(row, 'Height', 'Must be a whole number.', text));
   return null;
  }
  if (height < 24 || height > 88) {
   errors.push(createImportError(row, 'Height', 'Must be between 24 and 88 inches.', text));
   return null;
  }
  return height;
}

function validateYesNo(value, row, errors) {
  const text = String(value || '').trim().toLowerCase();
  if (!text) {
   errors.push(createImportError(row, 'Special Needs', 'Is required.', value));
   return null;
  }
  if (text !== 'y' && text !== 'yes' && text !== 'n' && text !== 'no') {
   errors.push(createImportError(row, 'Special Needs', 'Must be Y, Yes, N, or No.', value));
   return null;
  }
  return (text === 'y' || text === 'yes') ? 'Y' : 'N';
}

function buildImportRow(values, headerIndexMap, rowNumber) {
  const getValue = (key) => {
   const index = headerIndexMap.get(key);
   return index == null ? '' : values[index];
  };

  const errors = [];
  const firstName = validateRequiredText(getValue('firstName'), 'First Name', 64, rowNumber, errors);
  const lastName = validateRequiredText(getValue('lastName'), 'Last Name', 64, rowNumber, errors);
  const parentFirstName = validateOptionalText(getValue('parentFirstName'), 'Parent First Name', 64, rowNumber, errors);
  const parentLastName = validateOptionalText(getValue('parentLastName'), 'Parent Last Name', 64, rowNumber, errors);
  const parentEmail = validateEmail(getValue('parentEmail'), rowNumber, errors);
  const phone = validatePhone(getValue('phone'), rowNumber, errors);
  const dob = parseDob(getValue('dob'), rowNumber, errors);
  const associationNumber = validateAssociationNumber(getValue('associationNumber'), rowNumber, errors);
  const gender = validateGender(getValue('gender'), rowNumber, errors);
  const rank = validateRank(getValue('rank'), rowNumber, errors);
  const studio = validateRequiredText(getValue('studio'), 'Studio', 64, rowNumber, errors);
  const height = validateHeight(getValue('height'), rowNumber, errors);
  const specialNeeds = validateYesNo(getValue('specialNeeds'), rowNumber, errors);

  if (errors.length) {
   return { errors };
  }

  return {
   competitor: {
     id: genId(),
     sourceRow: rowNumber,
     firstName,
     lastName,
     fullName: `${firstName} ${lastName}`,
     parentFirstName,
     parentLastName,
     parentFullName: [parentFirstName, parentLastName].filter(Boolean).join(' '),
     parentEmail,
     phone,
     dob: dob.raw,
     dobIso: dob.iso,
     associationNumber,
     genderCode: gender === 'Male' ? 'M' : 'F',
     gender,
     rankCode: String(getValue('rank') || '').trim().toUpperCase(),
     rank,
     studio,
     school: studio,
     height,
     specialNeeds,
     specialNeedsFlag: specialNeeds === 'Y',
     age: dob.age,
   },
  };
}

function parseCSV(text) {
  const lines = text.trim().split(/\r?\n/).filter(l => l.trim());
  if (!lines.length) {
   return {
     competitors: [],
     errors: [createImportError('File', 'File', 'CSV is empty.', '')],
   };
  }

  const headerValues = parseCSVLine(lines[0]);
  const headerIndexMap = new Map();
  const duplicateHeaders = new Set();
  const normalizedHeaders = headerValues.map(normalizeImportHeader);

  headerValues.forEach((value, index) => {
   const normalized = normalizeImportHeader(value);
   const mapped = IMPORT_HEADER_ALIASES[normalized];
   if (!mapped) return;
   if (headerIndexMap.has(mapped)) {
     duplicateHeaders.add(mapped);
     return;
   }
   headerIndexMap.set(mapped, index);
  });

  const errors = [];
  const missingHeaders = IMPORT_HEADERS.filter((header) =>
   !OPTIONAL_IMPORT_HEADERS.has(header.key) && !headerIndexMap.has(header.key)
  );

  const legacyFormatDetected = normalizedHeaders.includes('full name') ||
   normalizedHeaders.includes('age') ||
   normalizedHeaders.includes('school') ||
   normalizedHeaders.includes('height (inches)');

  if (legacyFormatDetected) {
   errors.push(createImportError(
     'Header',
     'Header',
     'This CSV uses the old roster format. Please export the new registration format with First Name, Last Name, Email, Phone (optional), DOB, Association #, Gender, Current Rank, Studio, Special Needs, and Height.',
     ''
   ));
  }

  if (missingHeaders.length) {
   errors.push(createImportError(
     'Header',
     'Header',
     `Missing required column(s): ${missingHeaders.map((header) => header.label).join(', ')}.`,
     ''
   ));
  }

  if (duplicateHeaders.size) {
   errors.push(createImportError(
     'Header',
     'Header',
     `Duplicate column(s): ${Array.from(duplicateHeaders).map((key) => {
       const header = IMPORT_HEADERS.find((item) => item.key === key);
       return header ? header.label : key;
     }).join(', ')}.`,
     ''
   ));
  }

  if (lines.length === 1) {
   errors.push(createImportError('File', 'File', 'No competitor rows found.', ''));
  }

  const competitors = [];
  const rowErrors = [];

  for (let i = 1; i < lines.length; i += 1) {
   const values = parseCSVLine(lines[i]);
   const rowNumber = i + 1;
   const result = buildImportRow(values, headerIndexMap, rowNumber);
   if (result.errors) {
     rowErrors.push(...result.errors);
   } else {
     competitors.push(result.competitor);
   }
  }

  return {
   competitors,
   errors: errors.concat(rowErrors),
  };
}

function lowerGupBandRankOrder(label) {
  const order = ['TTLD', '10th-9th Gup', '8th-7th Gup', '6th-5th Gup'];
  return order.indexOf(label);
}

function blackBeltBandRankOrder(label) {
  const order = ['CDB', 'Cho Dan-Sam Dan'];
  return order.indexOf(label);
}

function groupsInSameMergePool(a, b) {
  const aBand = a.rankBand || groupRankBand(a.competitors);
  const bBand = b.rankBand || groupRankBand(b.competitors);

  if (a.gender !== b.gender) return false;
  if (aBand === bBand) return true;

  const aLow = lowerGupBandRankOrder(aBand);
  const bLow = lowerGupBandRankOrder(bBand);
  if (aLow >= 0 && bLow >= 0) return Math.abs(aLow - bLow) <= 1;

  const aBb = blackBeltBandRankOrder(aBand);
  const bBb = blackBeltBandRankOrder(bBand);
  if (aBb >= 0 && bBb >= 0) return Math.abs(aBb - bBb) <= 1;

  return false;
}

function resolveOverlappingAgeWindows(groups, minSize, maxSize, ageSpanUnder14, ageSpan14To37, ageSpan38Plus) {
  const buckets = new Map();
  for (const g of groups) {
    const key = `${g.gender}|${g.rankBand || groupRankBand(g.competitors)}`;
    if (!buckets.has(key)) buckets.set(key, []);
    buckets.get(key).push(g);
  }

  const resolved = [];
  for (const bucket of buckets.values()) {
    bucket.sort((a, b) => {
      const aMin = Math.min(...a.competitors.map(c => c.age));
      const bMin = Math.min(...b.competitors.map(c => c.age));
      return aMin - bMin;
    });

    let i = 0;
    while (i < bucket.length) {
      const current = bucket[i];
      const currentAges = current.competitors.map(c => c.age);
      const currentMinAge = Math.min(...currentAges);
      const currentMaxAge = Math.max(...currentAges);
      const currentSpan = currentMaxAge - currentMinAge;
      const currentSpanLimit = currentMinAge < 14 ? ageSpanUnder14 : currentMinAge < 38 ? ageSpan14To37 : ageSpan38Plus;
      const currentFlexLimit = currentMinAge < 14 ? 5 : currentMinAge < 38 ? 14 : 35;

      const next = bucket[i + 1];
      if (!next) {
        resolved.push(current);
        break;
      }

      const combined = [...current.competitors, ...next.competitors];
      const ages = combined.map(c => c.age);
      const minAge = Math.min(...ages);
      const maxAge = Math.max(...ages);
      const span = maxAge - minAge;
      const spanLimit = minAge < 14 ? ageSpanUnder14 : minAge < 38 ? ageSpan14To37 : ageSpan38Plus;
      const flexLimit = minAge < 14 ? 5 : minAge < 38 ? 14 : 35;
      const canMerge = combined.length <= maxSize && (span <= spanLimit || span <= flexLimit) && rankWindowAllowed(combined.map(c => c.rank));

      if (canMerge) {
        current.competitors = combined;
        current.isAdult = current.competitors.some(c => isAdult(c.age));
        current.rankBand = groupRankBand(current.competitors) || current.rankBand;
        current.name = `${current.gender} ${current.rankBand} · ${groupAgeRange(current.competitors)}`;
        bucket.splice(i + 1, 1);
        continue;
      }

      resolved.push(current);
      i += 1;
    }
  }

  return resolved;
}

function mergeCompatibleGroups(groups, minSize, maxSize, ageSpanUnder14, ageSpan14To37, ageSpan38Plus) {
  let current = [...groups];

  for (let pass = 0; pass < 3; pass++) {
    let changed = true;
    while (changed) {
      changed = false;
      for (let i = 0; i < current.length; i++) {
        const a = current[i];
        for (let j = i + 1; j < current.length; j++) {
          const b = current[j];

          if (a.gender !== b.gender) continue;

          const sameBand = a.rankBand === b.rankBand;
          const adjacentPool = groupsInSameMergePool(a, b);
          const broadSameGender = pass === 2;

          if (pass === 0 && !sameBand) continue;
          if (pass === 1 && !adjacentPool) continue;
          if (pass === 2 && !broadSameGender) continue;

          const combined = [...a.competitors, ...b.competitors];
          if (combined.length > maxSize) continue;

          const ages = combined.map(c => c.age);
          const minAge = Math.min(...ages);
          const maxAge = Math.max(...ages);
          const span = maxAge - minAge;
          const spanLimit = minAge < 14 ? ageSpanUnder14 : minAge < 38 ? ageSpan14To37 : ageSpan38Plus;
          const flexLimit = minAge < 14 ? 5 : minAge < 38 ? 14 : 35;
          const allowedByAge = 
            span <= spanLimit ||
            (span <= flexLimit && (combined.length < minSize || combined.length <= maxSize));
          const allowedByRank = rankWindowAllowed(combined.map(c => c.rank));

          if (allowedByAge && allowedByRank) {
            current.splice(j, 1);
            a.competitors = combined;
            a.isAdult = a.competitors.some(c => isAdult(c.age));
            a.rankBand = groupRankBand(a.competitors) || a.rankBand;
            a.name = `${a.gender} ${a.rankBand} · ${groupAgeRange(a.competitors)}`;
            changed = true;
            break;
          }
        }
        if (changed) break;
      }
    }
  }

  return current;
}

function runAutoGroupAlgo(competitors, settings) {
  const { minSize, maxSize, ageSpanUnder14, ageSpan14To37, ageSpan38Plus } = settings;
  const groups = [];
  const ttldComps = competitors.filter((c) => isTtldRank(c.rank));

  if (ttldComps.length) {
    const ttldGroups = [];
    const ttldBuckets = [
      ttldComps.filter((c) => (c.gender || 'Unknown') === 'Male'),
      ttldComps.filter((c) => (c.gender || 'Unknown') === 'Female'),
      ttldComps.filter((c) => (c.gender || 'Unknown') === 'Unknown'),
    ];

    for (const bucket of ttldBuckets) {
      if (!bucket.length) continue;
      const sorted = [...bucket].sort((a, b) =>
        a.age !== b.age ? a.age - b.age : rankIdx(a.rank) - rankIdx(b.rank)
      );

      let remaining = [...sorted];
      const rawGroups = [];

      while (remaining.length) {
        const seed = remaining.shift();
        const grp = [seed];
        let minAge = seed.age;
        let maxAge = seed.age;
        const spanLimit = seed.age < 14 ? ageSpanUnder14 : seed.age < 38 ? ageSpan14To37 : ageSpan38Plus;
        const flexLimit = seed.age < 14 ? 5 : seed.age < 38 ? 14 : 35;

        const leftover = [];
        for (const c of remaining) {
          const newMinAge = Math.min(minAge, c.age);
          const newMaxAge = Math.max(maxAge, c.age);
          const span = newMaxAge - newMinAge;
          const allowedByAge = span <= spanLimit || (grp.length < minSize && span <= flexLimit);
          const allowed = allowedByAge && grp.length < maxSize;
          if (allowed) {
            grp.push(c);
            minAge = newMinAge;
            maxAge = newMaxAge;
          } else {
            leftover.push(c);
          }
        }
        remaining = leftover;
        rawGroups.push(grp);
      }

      ttldGroups.push(...rawGroups);
    }

    let changed = true;
    while (changed) {
      changed = false;
      for (let i = 0; i < ttldGroups.length; i++) {
        let bestJ = -1;
        for (let j = 0; j < ttldGroups.length; j++) {
          if (i === j) continue;
          const combined = [...ttldGroups[i], ...ttldGroups[j]];
          if (combined.length > maxSize) continue;
          const ages = combined.map((c) => c.age);
          const minAge = Math.min(...ages);
          const maxAge = Math.max(...ages);
          const span = maxAge - minAge;
          const spanLimit = minAge < 14 ? ageSpanUnder14 : minAge < 38 ? ageSpan14To37 : ageSpan38Plus;
          const flexLimit = minAge < 14 ? 5 : minAge < 38 ? 14 : 35;
          const allowedByAge =
            span <= spanLimit ||
            (combined.length < minSize && span <= flexLimit) ||
            (combined.length >= minSize && combined.length <= maxSize && span <= flexLimit);
          if (allowedByAge) {
            bestJ = j;
            break;
          }
        }

        if (bestJ !== -1) {
          ttldGroups[bestJ] = [...ttldGroups[bestJ], ...ttldGroups[i]];
          ttldGroups.splice(i, 1);
          changed = true;
          break;
        }
      }
    }

    for (const grp of ttldGroups) {
      const ages = grp.map((c) => c.age);
      const minAge = Math.min(...ages);
      const maxAge = Math.max(...ages);
      const spanLabel = minAge === maxAge ? `Age ${minAge}` : `Ages ${minAge}-${maxAge}`;
      groups.push({
        id: genId(),
        competitors: grp,
        isAdult: grp.some((c) => isAdult(c.age)),
        gender: groupGender(grp),
        rankBand: 'TTLD',
        name: `${groupGender(grp)} ${groupRankRange(grp)} · ${spanLabel}`,
      });
    }
  }

  for (const gender of ['Male', 'Female', 'Unknown']) {
    const genderComps = competitors.filter(c => (c.gender || 'Unknown') === gender && !isTtldRank(c.rank));
    if (!genderComps.length) continue;

    for (const band of RANK_BANDS) {
      const bandComps = genderComps.filter(c => {
        const cBand = rankBandFor(c.rank);
        return cBand && cBand.key === band.key;
      });
      if (!bandComps.length) continue;

      const sorted = [...bandComps].sort((a, b) =>
        a.age !== b.age ? a.age - b.age : rankIdx(a.rank) - rankIdx(b.rank)
      );

      let remaining = [...sorted];
      const rawGroups = [];

      while (remaining.length) {
        const seed = remaining.shift();
        const grp = [seed];
        let minAge = seed.age;
        let maxAge = seed.age;
        const spanLimit = seed.age < 14 ? ageSpanUnder14 : seed.age < 38 ? ageSpan14To37 : ageSpan38Plus;
        const flexLimit = seed.age < 14 ? 5 : seed.age < 38 ? 14 : 35;

        const leftover = [];
        for (const c of remaining) {
          const newMinAge = Math.min(minAge, c.age);
          const newMaxAge = Math.max(maxAge, c.age);
          const span = newMaxAge - newMinAge;
          const allowedByAge = span <= spanLimit || (grp.length < minSize && span <= flexLimit);
          const allowedByRank = rankWindowAllowed([...grp.map(o => o.rank), c.rank]);
          const allowed = allowedByAge && allowedByRank && grp.length < maxSize;
          if (allowed) {
            grp.push(c);
            minAge = newMinAge;
            maxAge = newMaxAge;
          } else {
            leftover.push(c);
          }
        }
        remaining = leftover;
        rawGroups.push(grp);
      }

      let changed = true;
      while (changed) {
        changed = false;
        for (let i = 0; i < rawGroups.length; i++) {
          let bestJ = -1;
          for (let j = 0; j < rawGroups.length; j++) {
            if (i === j) continue;
            const combined = [...rawGroups[i], ...rawGroups[j]];
            if (combined.length > maxSize) continue;
            const ages = combined.map(c => c.age);
            const minAge = Math.min(...ages);
            const maxAge = Math.max(...ages);
            const span = maxAge - minAge;
            const spanLimit = minAge < 14 ? ageSpanUnder14 : minAge < 38 ? ageSpan14To37 : ageSpan38Plus;
            const flexLimit = minAge < 14 ? 5 : minAge < 38 ? 14 : 35;
            const allowedByAge =
              span <= spanLimit ||
              (combined.length < minSize && span <= flexLimit) ||
              (combined.length >= minSize && combined.length <= maxSize && span <= flexLimit);
            const allowedByRank = rankWindowAllowed(combined.map(c => c.rank));
            if (allowedByAge && allowedByRank) {
              bestJ = j;
              break;
            }
          }

          if (bestJ !== -1) {
            rawGroups[bestJ] = [...rawGroups[bestJ], ...rawGroups[i]];
            rawGroups.splice(i, 1);
            changed = true;
            break;
          }
        }
      }

      for (const grp of rawGroups) {
        const ages = grp.map(c => c.age);
        const minAge = Math.min(...ages);
        const maxAge = Math.max(...ages);
        const spanLabel = minAge === maxAge ? `Age ${minAge}` : `Ages ${minAge}-${maxAge}`;
        groups.push({
          id: genId(),
          competitors: grp,
          isAdult: grp.some(c => isAdult(c.age)),
          gender,
          rankBand: band.label,
          name: `${gender} ${groupRankRange(grp)} · ${spanLabel}`,
        });
      }
    }
  }

  const rebalanced = resolveOverlappingAgeWindows(groups, minSize, maxSize, ageSpanUnder14, ageSpan14To37, ageSpan38Plus);
  const merged = mergeCompatibleGroups(rebalanced, minSize, maxSize, ageSpanUnder14, ageSpan14To37, ageSpan38Plus);
  merged.forEach(g => { if (!g.name) g.name = autoName(g); });
  return merged;
}

function setupFileHandlers() {
  const zone  = document.getElementById('dropZone');
  const input = document.getElementById('fileInput');

  zone.addEventListener('click', () => input.click());
  zone.addEventListener('dragover', e => { e.preventDefault(); zone.classList.add('drag-over'); });
  zone.addEventListener('dragleave', () => zone.classList.remove('drag-over'));
  zone.addEventListener('drop', e => {
    e.preventDefault();
    zone.classList.remove('drag-over');
    if (e.dataTransfer.files[0]) loadFile(e.dataTransfer.files[0]);
  });
  input.addEventListener('change', e => { if (e.target.files[0]) loadFile(e.target.files[0]); });
}

function loadFile(file) {
  const reader = new FileReader();
  reader.onload = e => {
   try {
     const result = parseCSV(e.target.result);
     state.competitors = result.competitors;
     state.groups = [];
     state.unassigned = [];
     renderImportPreview(result);
     document.getElementById('reviewSection').classList.add('hidden');
   } catch (error) {
     renderImportParseError(error);
   }
  };
  reader.readAsText(file);
}

function renderImportParseError(error) {
  const message = error instanceof Error ? error.message : String(error || 'Unknown CSV parsing error');
  const validationSection = document.getElementById('importValidation');
  const validationBody = document.getElementById('validationBody');
  const validationSummary = document.getElementById('validationSummary');

  document.getElementById('importPreview').classList.add('hidden');
  document.getElementById('reviewSection').classList.add('hidden');
  document.getElementById('autoGroupBtn').disabled = true;
  document.getElementById('adultNote').style.display = 'none';

  validationSection.classList.remove('hidden');
  validationSummary.textContent = `Unable to parse the CSV file: ${message}`;
  validationBody.innerHTML = '';
}

function renderImportPreview(result) {
  const comps = result.competitors || [];
  const errors = result.errors || [];
  const hasErrors = errors.length > 0;
  document.getElementById('importPreview').classList.remove('hidden');
  document.getElementById('adultNote').style.display = hasErrors ? 'none' : 'block';
  document.getElementById('autoGroupBtn').disabled = !comps.length;

  applyRecommendedSizeSettings(comps.length);

  if (hasErrors) {
   document.getElementById('importCount').textContent =
     `${comps.length} valid competitor${comps.length !== 1 ? 's' : ''} found; ${errors.length} issue${errors.length !== 1 ? 's' : ''} must be fixed first`;
  } else {
   document.getElementById('importCount').textContent =
     `${comps.length} competitor${comps.length !== 1 ? 's' : ''} imported`;
  }

  const show = comps.slice(0, 12);
  const more = comps.length > 12;
  document.getElementById('previewBody').innerHTML =
   show.map(c => `
     <tr>
       <td>${esc(c.fullName)}</td>
       <td>${rankBadge(c.rank)}</td>
       <td>${c.age}</td>
       <td>${esc(c.height)}</td>
     </tr>
   `).join('') +
   (more ? `<tr><td colspan="4" style="text-align:center;color:#999;font-style:italic;padding:10px;">
     … and ${comps.length - 12} more competitors
   </td></tr>` : '');

  const validationSection = document.getElementById('importValidation');
  const validationBody = document.getElementById('validationBody');
  const validationSummary = document.getElementById('validationSummary');

  if (hasErrors) {
   validationSection.classList.remove('hidden');
   validationSummary.textContent = `${errors.length} issue${errors.length !== 1 ? 's' : ''} found during CSV validation. Fix the rows below and import again.`;
   validationBody.innerHTML = errors.map((error) => `
     <tr>
       <td>${esc(error.row)}</td>
       <td>${esc(error.field)}</td>
       <td>${esc(error.issue)}</td>
       <td>${esc(error.value)}</td>
     </tr>
   `).join('');
  } else {
   validationSection.classList.add('hidden');
   validationBody.innerHTML = '';
   validationSummary.textContent = '';
  }
}

function clearImport() {
  state = { competitors: [], groups: [], unassigned: [] };
  document.getElementById('importPreview').classList.add('hidden');
  document.getElementById('importValidation').classList.add('hidden');
  document.getElementById('reviewSection').classList.add('hidden');
  document.getElementById('autoGroupBtn').disabled = true;
  document.getElementById('adultNote').style.display = 'none';
  document.getElementById('sizeNote').style.display = 'none';
  document.getElementById('fileInput').value = '';
}

function runAutoGroup() {
  if (!state.competitors.length) return;

  const settings = {
    minSize: Math.max(MIN_GROUP_SIZE, parseInt(document.getElementById('minSize').value) || MIN_GROUP_SIZE),
    maxSize: parseInt(document.getElementById('maxSize').value)   || recommendedMaxSize(state.competitors.length),
    ageSpanUnder14: parseInt(document.getElementById('ageSpanUnder14').value) || 3,
    ageSpan14To37: parseInt(document.getElementById('ageSpan14To37').value) || 4,
    ageSpan38Plus: parseInt(document.getElementById('ageSpan38Plus').value) || 35,
  };

  state.groups     = runAutoGroupAlgo(state.competitors, settings);
  state.unassigned = [];

  document.getElementById('reviewSection').classList.remove('hidden');
  renderReview();
}

function renderReview() {
  state.groups.forEach(syncGroupDisplayMetadata);
  renderStats();
  renderUnassigned();
  renderGroups();
}

function renderStats() {
  const minSize    = Math.max(MIN_GROUP_SIZE, parseInt(document.getElementById('minSize').value || MIN_GROUP_SIZE));
  const total      = state.groups.reduce((s, g) => s + g.competitors.length, 0) + state.unassigned.length;
  const nGroups    = state.groups.length;
  const nWarn      = state.groups.filter(g => g.competitors.length < minSize).length;
  const nAdults    = state.groups.filter(g => g.isAdult).reduce((s, g) => s + g.competitors.length, 0);
  const nJuniors   = total - nAdults - state.unassigned.length;

  document.getElementById('statsBar').innerHTML = `
    <div class="stat">
      <span class="stat-value">${total}</span>
      <span class="stat-label">Competitors</span>
    </div>
    <div class="stat">
      <span class="stat-value">${nGroups}</span>
      <span class="stat-label">Groups</span>
    </div>
    <div class="stat">
      <span class="stat-value">${nAdults}</span>
      <span class="stat-label">Adults</span>
    </div>
    <div class="stat">
      <span class="stat-value">${nJuniors}</span>
      <span class="stat-label">Juniors</span>
    </div>
    ${nWarn ? `<div class="stat stat-warn">
      <span class="stat-value">⚠ ${nWarn}</span>
      <span class="stat-label">Undersized</span>
    </div>` : ''}
    ${state.unassigned.length ? `<div class="stat stat-warn">
      <span class="stat-value">⚠ ${state.unassigned.length}</span>
      <span class="stat-label">Unassigned</span>
    </div>` : ''}
  `;
}

function renderUnassigned() {
  const sec  = document.getElementById('unassignedSection');
  const list = document.getElementById('unassignedList');

  if (!state.unassigned.length) { sec.classList.add('hidden'); return; }
  sec.classList.remove('hidden');

  list.innerHTML = state.unassigned.map(c => `
    <div class="competitor-row pool-row"
         draggable="true"
         ondragstart="startCompetitorDrag(event, '${c.id}', null)"
         ondragend="endCompetitorDrag(event)">
      <div class="competitor-info">
        <div class="competitor-name">${esc(c.fullName)}</div>
        <div class="competitor-sub">${rankBadge(c.rank)}<span>Age ${c.age}</span></div>
      </div>
    </div>
  `).join('');
}

function renderGroups() {
  const minSize = Math.max(MIN_GROUP_SIZE, parseInt(document.getElementById('minSize').value || MIN_GROUP_SIZE));
  const grid = document.getElementById('groupsGrid');

  grid.innerHTML = state.groups.map((g, idx) => {
    const count = g.competitors.length;
    const warn  = count < minSize;

    return `
      <div class="group-card ${warn ? 'warn-border' : 'ok-border'}"
           id="gc-${g.id}"
           ondragover="allowGroupDrop(event, '${g.id}')"
           ondragenter="markGroupDropTarget(event, '${g.id}')"
           ondragleave="unmarkGroupDropTarget(event, '${g.id}')"
           ondrop="dropIntoGroup(event, '${g.id}')">
        <div class="group-header ${g.isAdult ? 'adult' : ''}">
          <input class="group-name-input"
                 value="${esc(g.name)}"
                 readonly
                 title="Auto-generated from current competitors">
          <div class="group-meta">
            <span>${count} competitor${count !== 1 ? 's' : ''}</span>
            <span>${groupRankRange(g.competitors)}</span>
            <span>${groupAgeRange(g.competitors)}</span>
          </div>
        </div>
        ${warn ? `<div class="group-warn-strip">⚠️ Only ${count} competitor${count !== 1 ? 's' : ''} — minimum is ${minSize}</div>` : ''}
        <div class="competitor-list">
          ${g.competitors.map(c => competitorRow(c, g.id)).join('')}
        </div>
        <div class="group-footer">
          <span style="font-size:12px;color:#999;">Group ${idx + 1}</span>
        </div>
      </div>
    `;
  }).join('');
}

function competitorRow(c, groupId) {
  return `
    <div class="competitor-row"
         draggable="true"
         ondragstart="startCompetitorDrag(event, '${c.id}', '${groupId}')"
         ondragend="endCompetitorDrag(event)">
      <div class="competitor-info">
        <div class="competitor-name">${esc(c.fullName)}</div>
        <div class="competitor-sub">${rankBadge(c.rank)}<span>Age ${c.age}</span></div>
      </div>
      <div class="competitor-actions">
        <button class="btn btn-danger btn-sm" onclick="unassign('${c.id}','${groupId}')">✕</button>
      </div>
    </div>
  `;
}

function rankBadge(rank) {
  if (rankIdx(rank) === -1) return `<span class="rank-badge rank-unk">? ${esc(rank)}</span>`;
  const cls = isBlackBelt(rank) ? 'rank-badge rank-black' : 'rank-badge rank-gup';
  return `<span class="${cls}">${esc(rank)}</span>`;
}

function renameGroup(id, name) {
  const g = state.groups.find(g => g.id === id);
  if (g) g.name = name;
}

function deleteGroup(id) {
  const g = state.groups.find(g => g.id === id);
  if (!g) return;
  if (!confirm(`Delete "${g.name}"? Its ${g.competitors.length} competitor(s) will move to Unassigned.`)) return;
  state.unassigned.push(...g.competitors);
  state.groups = state.groups.filter(g => g.id !== id);
  renderReview();
}

function addNewGroup() {
  state.groups.push({ id: genId(), competitors: [], isAdult: false, name: 'New Group' });
  renderReview();
  setTimeout(() => window.scrollTo({ top: document.body.scrollHeight, behavior: 'smooth' }), 80);
}

function unassign(compId, groupId) {
  const g = state.groups.find(g => g.id === groupId);
  if (!g) return;
  const idx = g.competitors.findIndex(c => c.id === compId);
  if (idx === -1) return;
  state.unassigned.push(...g.competitors.splice(idx, 1));
  promptToDeleteEmptyGroup(g);
  renderReview();
}

function promptToDeleteEmptyGroup(group) {
  if (!group || group.competitors.length !== 0) return;
  const shouldDelete = confirm(`"${group.name}" is now empty. Delete this group?`);
  if (!shouldDelete) return;
  state.groups = state.groups.filter((candidate) => candidate.id !== group.id);
}

function moveCompetitorBetweenGroups(compId, fromGroupId, targetGroupId) {
  if (!targetGroupId || fromGroupId === targetGroupId) return false;

  const sourceGroup = fromGroupId ? state.groups.find((g) => g.id === fromGroupId) : null;
  const sourceList = sourceGroup ? sourceGroup.competitors : state.unassigned;
  const sourceIndex = sourceList.findIndex((c) => c.id === compId);
  if (sourceIndex === -1) return false;
  const competitor = sourceList[sourceIndex];

  const targetGroup = state.groups.find((g) => g.id === targetGroupId);
  if (!targetGroup) return false;
  if (!canAcceptMove(targetGroup, competitor)) return false;

  sourceList.splice(sourceIndex, 1);
  targetGroup.competitors.push(competitor);
  promptToDeleteEmptyGroup(sourceGroup);
  renderReview();
  return true;
}

function canAcceptMove(group, competitor) {
  if (!group || !competitor) return false;

  if (isTtldRank(competitor.rank)) {
    return group.competitors.length === 0 || groupIsTtldOnly(group);
  }

  return !groupIsTtldOnly(group);
}

function startCompetitorDrag(event, compId, fromGroupId) {
  dragCtx = { compId, fromGroupId };
  event.dataTransfer.effectAllowed = 'move';
  event.dataTransfer.setData('text/plain', JSON.stringify(dragCtx));
}

function endCompetitorDrag(event) {
  dragCtx = null;
  document.querySelectorAll('.group-card.drop-target').forEach((el) => el.classList.remove('drop-target'));
}

function allowGroupDrop(event, groupId) {
  if (!dragCtx) return;
  if (dragCtx.fromGroupId === groupId) return;
  const target = state.groups.find((group) => group.id === groupId);
  const source = dragCtx.fromGroupId
    ? state.groups.find((group) => group.id === dragCtx.fromGroupId)
    : null;
  const sourceList = source ? source.competitors : state.unassigned;
  const competitor = sourceList.find((c) => c.id === dragCtx.compId);
  if (!target || !competitor || !canAcceptMove(target, competitor)) return;
  event.preventDefault();
  event.dataTransfer.dropEffect = 'move';
}

function markGroupDropTarget(event, groupId) {
  if (!dragCtx || dragCtx.fromGroupId === groupId) return;
  const card = document.getElementById(`gc-${groupId}`);
  if (card) card.classList.add('drop-target');
}

function unmarkGroupDropTarget(event, groupId) {
  const card = document.getElementById(`gc-${groupId}`);
  if (card) card.classList.remove('drop-target');
}

function dropIntoGroup(event, targetGroupId) {
  event.preventDefault();
  const card = document.getElementById(`gc-${targetGroupId}`);
  if (card) card.classList.remove('drop-target');

  let payload = dragCtx;
  if (!payload && event.dataTransfer) {
    try {
      payload = JSON.parse(event.dataTransfer.getData('text/plain') || 'null');
    } catch (error) {
      payload = null;
    }
  }

  if (!payload) return;
  moveCompetitorBetweenGroups(payload.compId, payload.fromGroupId, targetGroupId);
  dragCtx = null;
}

          phone: c.phone,
          dob: c.dob,
          dobIso: c.dobIso,
          associationNumber: c.associationNumber,
          genderCode: c.genderCode,
          rank: c.rank,
          rankCode: c.rankCode,
          age: c.age,
          studio: c.studio,
          school: c.school,
          height: c.height,
          specialNeeds: c.specialNeeds,
          specialNeedsFlag: c.specialNeedsFlag,
        })),
      };
    }),
    ...(state.unassigned.length ? {
      unassigned: state.unassigned.map(c => ({
        firstName: c.firstName,
        lastName: c.lastName,

        parentFirstName: c.parentFirstName,
